\begin{block}{\texttt{models.py} 完整代码}
\footnotesize
\begin{lstlisting}[basicstyle=\ttfamily\scriptsize, frame=none, numbers=none, backgroundcolor=\color{codebg}]
class MLPClassifier(nn.Module):
    def __init__(self):
        super().__init__()
        self.net = nn.Sequential(
            nn.Flatten(),        # (a)
            nn.Linear(784, 128), # (b)
            nn.ReLU(),           # (c)
            nn.Linear(128, 64),  # (d)
            nn.ReLU(),           # (e)
            nn.Linear(64, 10),   # (f)
        )
    def forward(self, x):
        return self.net(x)       # (g)
\end{lstlisting}
\end{block}
\vspace{0.1cm}
\footnotesize
\textbf{标注对应后面各页详解：}
\begin{itemize}
  \item (a) Flatten：把图像"拉直"成一维向量
  \item (b)(d)(f) Linear：矩阵乘法 $y=Wx+b$，共 3 层
  \item (c)(e) ReLU：$\max(0,x)$，引入非线性
  \item (g) forward：一行代码完成全部前向计算
\end{itemize}
\end{columns}
\end{frame}

% ═══════════ 5 — Flatten + Linear + ReLU 直观解释 ═══════════
\begin{frame}[shrink=15]{从像素到特征：Flatten、Linear、ReLU 各做了什么}
\small

\begin{columns}[T]
\column{0.33\textwidth}
\begin{block}{Flatten（\texttt{models.py} (a)）}
\begin{center}
\includegraphics[width=2.5cm]{slide_assets/digit_grid.png}
\end{center}
\vspace{-0.3cm}
\footnotesize
把 28$\times$28 的矩阵\\
"拉直"成一维向量。\\
\textbf{不改变数值}，\\
只是重新排列。

\textbf{代价：}丢弃像素间的\\
空间位置关系。这正是\\
MLP 与 CNN 的核心区别。
\end{block}

\column{0.33\textwidth}
\begin{block}{Linear（\texttt{models.py} (b)(d)(f)）}
\footnotesize
\[
y = Wx + b
\]

\vspace{-0.2cm}
每个 Linear 层是一个\\
"特征检测器"。

$W$ 的每一行 = 一个\\
"模式模板"：\\
\hspace{4pt}$w_{ij}>0$：像素 $j$ 推高输出 $i$\\
\hspace{4pt}$w_{ij}<0$：像素 $j$ 拉低输出 $i$

$b$：允许输出整体平移。
\end{block}

\column{0.34\textwidth}
\begin{block}{ReLU（\texttt{models.py} (c)(e)）}
\footnotesize
\[
f(x) = \max(0, x)
\]
\begin{center}
\begin{tikzpicture}[scale=1.1]
  \draw[->, gray] (-1.2,0) -- (1.2,0);
  \draw[->, gray] (0,-0.3) -- (0,1.3);
  \draw[ultra thick, mainblue] (-1.2,0) -- (0,0) -- (1.2,1.2);
\end{tikzpicture}
\end{center}
\vspace{-0.3cm}
\textbf{为什么必须？}\\
没有 ReLU，两层 Linear\\
等价于一层：\\
$W_2(W_1 x+b_1)+b_2$\\
$=W'x + b'$

ReLU 打破"线性坍缩"。
\end{block}
\end{columns}
\end{frame}

% ═══════════ 6 — Softmax 直观解释 ═══════════
\begin{frame}[shrink=25]{Softmax：把原始分数变成概率分布}
\small

\begin{columns}[T]
\column{0.48\textwidth}
\begin{center}
\includegraphics[width=\textwidth]{slide_assets/softmax_viz.png}
\end{center}
\vspace{-0.2cm}
\begin{exampleblock}{如何读左图？}
\footnotesize
\textbf{左图 Logits：} 模型原始输出，可正可负，数值范围不固定。类别 7 的分数 2.8 最高，但难以直观判断"有多大把握"。\\[2pt]
\textbf{右图 Softmax 后：} 转为概率——类别 7 概率约 0.62，其余 9 类合计约 0.38。一眼看出模型对"7"有 62\% 把握。\\[2pt]
\textbf{指数 $e^{z_i}$ 的效果：} 放大高分、缩小低分，让最大 logit 在概率中\"脱颖而出\"。
\end{exampleblock}

\column{0.52\textwidth}
\begin{block}{Softmax 做了什么？}
\footnotesize
模型输出 10 个 \textbf{logits}（原始分数，可正可负）。Softmax 把它变成概率分布（非负、和为 1）：
\[
p_i = \frac{e^{z_i}}{\sum_{j=0}^{9} e^{z_j}}
\]
\end{block}

\vspace{0.1cm}

\begin{exampleblock}{直观类比}
\footnotesize
Logits = 评委打分，Softmax = 按分数分配获奖概率。\\
分数最高者概率最高，但其他人也有机会。
\end{exampleblock}

\vspace{0.1cm}

\footnotesize
\textbf{为何模型里不加 Softmax？} CrossEntropyLoss 内部已包含。\\[1pt]
\textbf{代码：} \texttt{models.py} Linear(64,10) 出 logits，\texttt{mnist\_mlp.py:74} CrossEntropyLoss 自动做 Softmax。
\end{columns}
\end{frame}

% ═══════════ 7 — 交叉熵损失直观解释 ═══════════
\begin{frame}[fragile,shrink=27]{交叉熵损失 (CrossEntropyLoss)：量化"预测有多偏离正确答案"}
\small

\begin{center}
\includegraphics[width=0.85\textwidth]{slide_assets/crossentropy_viz.png}
\end{center}

\vspace{0.15cm}

\begin{columns}[T]
\column{0.5\textwidth}
\begin{block}{数学定义（\texttt{mnist\_mlp.py:74}）}
\footnotesize
\begin{lstlisting}[basicstyle=\ttfamily\footnotesize, frame=none, numbers=none, backgroundcolor=\color{codebg}]
criterion = nn.CrossEntropyLoss()
\end{lstlisting}
内部两步：\\[2pt]
1. \textbf{Softmax}：$p_i = e^{z_i} / \sum e^{z_j}$\\[2pt]
2. \textbf{负对数似然}：
   $\text{Loss} = -\ln(p_{\text{true}})$
\end{block}

\column{0.5\textwidth}
\begin{exampleblock}{直观理解：三个场景}
\footnotesize
\textbf{场景 1 (Loss=0.16)：} 模型很确信，而且对了——Loss 很小。

\textbf{场景 2 (Loss=1.05)：} 模型不太确定，概率分散到几个类别——Loss 中等。

\textbf{场景 3 (Loss=2.99)：} 模型非常确信一个错误的答案——Loss 最大，这是最糟糕的情况！
\end{exampleblock}
\end{columns}

\vspace{0.15cm}
\footnotesize
\textbf{在训练循环中：} \texttt{mnist\_mlp.py:29} 的 \texttt{loss = criterion(logits, targets)} 计算的就是这个值。
\end{frame}

% ═══════════ 9 — 训练循环核心 5 步 ═══════════
\begin{frame}[fragile,shrink=20]{训练循环：\texttt{mnist\_mlp.py} 中 \texttt{train\_one\_epoch} 的 5 个步骤}
\footnotesize

\begin{block}{核心代码（\texttt{mnist\_mlp.py:24--31}）}
\begin{lstlisting}[basicstyle=\ttfamily\small, frame=single, backgroundcolor=\color{codebg}]
for images, targets in train_loader:
    images, targets = images.to(device), targets.to(device)

    optimizer.zero_grad()                  # 1. 清空累积的梯度
    logits = model(images)                 # 2. 前向传播 → 预测分数
    loss = criterion(logits, targets)      # 3. 计算损失
    loss.backward()                        # 4. 反向传播 → 计算梯度
    optimizer.step()                       # 5. 参数更新 → 梯度下降
\end{lstlisting}
\end{block}

% ═══════════ 10 — 反向传播 + 梯度下降 ═══════════
\begin{frame}[fragile,shrink=20]{梯度下降：模型如何"学习"——在损失平面上寻找最低点}
\small

\begin{columns}[T]
\column{0.5\textwidth}
\begin{center}
\includegraphics[width=\textwidth]{slide_assets/gd_landscape.png}
\end{center}

\column{0.5\textwidth}
\begin{block}{反向传播（\texttt{mnist\_mlp.py:30}）}
\footnotesize
\texttt{loss.backward()} 自动计算每个参数的梯度：
\[
\frac{\partial L}{\partial w} \quad\text{对所有 } w
\]
从损失出发，沿计算图反向逐层用链式法则求偏导数。这就是 PyTorch \textbf{autograd} 的核心。
\end{block}

\vspace{0.15cm}

\begin{block}{梯度下降（\texttt{mnist\_mlp.py:31 + :75}）}
\footnotesize
\texttt{optimizer.step()} 沿梯度负方向更新参数：
\[
w_{\text{new}} = w_{\text{old}} - \eta \cdot \frac{\partial L}{\partial w}
\]
\begin{lstlisting}[basicstyle=\ttfamily\footnotesize, frame=none, numbers=none, backgroundcolor=\color{codebg}]
optimizer = torch.optim.Adam(
    model.parameters(), lr=1e-3)
\end{lstlisting}
\textbf{Adam} = Momentum + RMSprop，自适应调整学习率。
\end{block}
\end{columns}

\vspace{0.15cm}
\footnotesize
\textbf{直观理解：} 想象你站在山顶（初始参数）要找最低点。你看一眼脚下的坡度（梯度），往最陡的下坡方向走一小步（步长 = 学习率 $\eta$）。重复这个过程，直到到达谷底。这就是梯度下降。
\end{frame}

\begin{block}{Batch GD（全量）}
\footnotesize
每次用全部 60,000 张图计算梯度。\\
\textbf{优点：} 梯度精确，路径平滑。\\
\textbf{缺点：} 每步太慢，显存放不下大数据集。
\end{block}

\column{0.33\textwidth}
\begin{block}{SGD（随机）}
\footnotesize
每次只用 1 张图。\\
\textbf{优点：} 极快，可跳出局部最优。\\
\textbf{缺点：} 噪声大，路径震荡。
\end{block}

\column{0.34\textwidth}
\begin{block}{Mini-batch GD（本实验用）}
\footnotesize
每次 64 张图（\texttt{data.py} \texttt{batch\_size}）。\\
\textbf{优点：} 兼顾速度和稳定性，GPU 并行高效。\\[1pt]
\textit{大多数深度学习都用 Mini-batch GD。}
\end{block}
\end{columns}
\end{frame}

\begin{frame}[fragile,shrink=15]{训练 vs 测试：\texttt{train\_one\_epoch} 与 \texttt{evaluate} 的对比}
\small

\begin{columns}[T]
\column{0.48\textwidth}
\begin{block}{\texttt{train\_one\_epoch}（\texttt{mnist\_mlp.py:12--36}）}
\footnotesize
\begin{lstlisting}[basicstyle=\ttfamily\footnotesize, frame=none, numbers=none, backgroundcolor=\color{codebg}]
model.train()          # 训练模式
for images, targets in loader:
    optimizer.zero_grad()
    logits = model(images)
    loss = criterion(logits, targets)
    loss.backward()     # 计算梯度
    optimizer.step()    # 更新参数
\end{lstlisting}
\alert{参数会改变！模型在"学习"。}
\end{block}

\column{0.52\textwidth}
\begin{block}{\texttt{evaluate}（\texttt{mnist\_mlp.py:39--56}）}
\footnotesize
\begin{lstlisting}[basicstyle=\ttfamily\footnotesize, frame=none, numbers=none, backgroundcolor=\color{codebg}]
@torch.no_grad()       # 关闭梯度
def evaluate(model, loader, ...):
    model.eval()        # 评估模式
    for images, targets in loader:
        logits = model(images)
        # 没有 backward()!
        # 没有 step()!
\end{lstlisting}
\alert{参数不变！只是使用已学到的知识。}
\end{block}
\end{columns}

\vspace{0.15cm}

\begin{exampleblock}{泛化：为什么要分开训练集和测试集？}
\footnotesize
模型可能通过"死记硬背"在训练集上得高分，但在新图片上表现糟糕（\textbf{过拟合}）。\\
测试集衡量的是模型的\textbf{泛化能力}——在"没见过的数据"上表现如何。\\
\begin{center}
训练准确率 $\approx$ 99\% \quad vs \quad 测试准确率 $\approx$ 97\% \quad $\to$ \quad 2\% 的差异 = 泛化误差
\end{center}
\end{exampleblock}

% ── 2.1 MLP 的局限 ──
\begin{frame}[shrink=20]{为什么 MLP 不够？——它丢弃了图像的空间结构}
\small

\begin{columns}[T]
\column{0.45\textwidth}
\begin{block}{MLP 的问题}
\footnotesize
\texttt{nn.Flatten()} 把 28$\times$28 图像 "拉直" 为 784 维向量。

\textbf{后果：}
\begin{itemize}
  \item 相邻像素和远距离像素被同等对待
  \item 左上角像素和右下角像素对第一个隐层神经元来说 \\ "距离一样远"
  \item 模型需要\textbf{浪费参数}去重新"发现"哪些像素是邻居
\end{itemize}

\begin{center}
\begin{tikzpicture}[scale=0.6]
  \fill[lightblue!20] (0,0) rectangle (3,3);
  \draw[step=0.5, gray] (0,0) grid (3,3);
  \draw[->, thick, redaccent] (3.3,1.5) -- (4.5,1.5);
  \fill[lightblue!50] (4.8,0.2) rectangle (7.5,2.8);
  \node at (6.15,1.5) {\small 784 维};
  \node at (1.5,-0.5) {28$\times$28 图像};
  \draw[redaccent, thick] (0.25,1.25) rectangle (0.75,1.75);
  \draw[redaccent, thick] (2.25,1.25) rectangle (2.75,1.75);
  \draw[redaccent, thick, dashed] (1.5,5.2) -- (1.5,3);
\end{tikzpicture}
\end{center}
\end{block}

\column{0.55\textwidth}
\begin{block}{CNN 的答案：保留二维结构}
\footnotesize
\textbf{核心思想：} 图像中相邻的像素是相关的——笔画、边缘、纹理都是局部模式。

CNN 用\textbf{卷积核} (convolution kernel) 在图像上滑动，每次只看一个局部窗口。同一个核在整张图上共享参数。

\vspace{0.15cm}
\textbf{两个关键创新：}
\begin{enumerate}
  \item \textbf{局部感受野 (Local Receptive Field):}\\
  每个神经元只看一小块区域（如 3$\times$3），而非全部 784 个像素
  \item \textbf{权重共享 (Weight Sharing):}\\
  同一个 3$\times$3 卷积核滑过整张图，参数量与图像尺寸无关
\end{enumerate}

\vspace{0.15cm}
\includegraphics[width=\textwidth]{slide_assets/conv_demo.png}
\end{block}
\end{columns}
\end{frame}

% ── 2.2 卷积操作直观 ──
\begin{frame}[shrink=20]{卷积操作：一个会 "滑动" 的特征检测器}
\small

\begin{columns}[T]
\column{0.48\textwidth}
\begin{center}
\includegraphics[width=0.98\textwidth]{slide_assets/conv_demo.png}
\end{center}

\column{0.52\textwidth}
\begin{block}{卷积 = 点积 + 滑动}
\footnotesize
\[
\text{output}[i,j] = \sum_{r=0}^{2}\sum_{c=0}^{2} \text{kernel}[r,c] \cdot \text{input}[i+r, j+c]
\]

\textbf{直观理解：}
\begin{itemize}
  \item 卷积核 = 一个 "模板" 或 "特征检测器"
  \item 在图像上逐位置滑动，每步计算核与图像局部区域的相似度
  \item 相似度高 $\to$ 输出大（该位置匹配这个模式）
  \item 相似度低 $\to$ 输出小（该位置不匹配）
\end{itemize}
\end{block}

\vspace{0.1cm}

\begin{exampleblock}{例子：垂直边缘检测核}
\footnotesize
左图卷积核 $\begin{bmatrix}-1&0&1\\-1&0&1\\-1&0&1\end{bmatrix}$ 在图像中间（有竖线的地方）输出大，在平坦区域输出小。\\[1pt]
\textbf{训练时，这些核的数值是自动学到的——不需要人工设计！}
\end{exampleblock}
\end{columns}
\end{frame}

% ── 2.3 Max Pooling ──
\begin{frame}[fragile,shrink=20]{Max Pooling（最大池化）：压缩信息、增强鲁棒性}
\small

\begin{columns}[T]
\column{0.45\textwidth}
\begin{block}{MaxPool2d(2) 做了什么？}
\footnotesize
把特征图划分为 2$\times$2 的小块，每块只保留最大值。

\begin{center}
\begin{tikzpicture}[scale=0.9]
  % 4x4 input
  \fill[lightblue!20] (0,0) rectangle (4,4);
  \draw[step=1, gray] (0,0) grid (4,4);

  \node at (2.5,2.5) {1}; \node at (3.5,2.5) {4};
  \node at (0.5,1.5) {2}; \node at (1.5,1.5) {1};
  \node at (2.5,1.5) {6}; \node at (3.5,1.5) {2};

  \node at (2,-0.3) {4$\times$4 输入};

  \draw[->, thick, mainblue] (4.3,2) -- (5.3,2);

  % 2x2 output
  \fill[lightblue!40] (5.5,1) rectangle (7.5,3);

\end{tikzpicture}
\end{center}
\end{block}

\column{0.55\textwidth}
\begin{block}{为什么用 MaxPool？}
\footnotesize
\textbf{三个作用：}
\begin{enumerate}
  \item \textbf{降采样：} 尺寸减半（28$\to$14$\to$7），减少计算量
  \item \textbf{增大感受野：} 后续层能看到更大的图像区域
  \item \textbf{平移不变性：} 图像稍微平移，最大值通常还在窗口内
\end{enumerate}

\vspace{0.15cm}
\textbf{在我们的 CNN 中（\texttt{models.py}）：}
\begin{lstlisting}[basicstyle=\ttfamily\footnotesize, frame=none, numbers=none, backgroundcolor=\color{codebg}]
nn.Conv2d(1, 16, 3, padding=1),  # 28x28
nn.ReLU(),
nn.MaxPool2d(2),                 # -> 14x14
nn.Conv2d(16, 32, 3, padding=1), # 14x14
nn.ReLU(),
nn.MaxPool2d(2),                 # -> 7x7
\end{lstlisting}
\end{block}

\vspace{0.15cm}
\footnotesize
\textbf{Padding=1 的作用：} 在图像边缘补一圈 0，使卷积后尺寸不变（28$\times$28 $\to$ 28$\times$28）。\\
不用 padding 的话，3$\times$3 卷积会使尺寸缩小 2 像素（28$\to$26）。
\end{columns}
\end{frame}

        super().__init__()
        self.net = nn.Sequential(
            nn.Conv2d(1,16,3,padding=1),
            nn.ReLU(),

% ── 3.3 C 推理代码 ──
\begin{frame}[fragile,shrink=29]{C 语言推理：\texttt{mnist\_mlp\_infer.c} 核心函数}
\footnotesize

\begin{columns}[T]
\column{0.5\textwidth}
\begin{block}{矩阵乘向量 + 偏置 (\texttt{linear})}
\begin{lstlisting}[basicstyle=\ttfamily\footnotesize, frame=none, numbers=none, backgroundcolor=\color{codebg}]
void linear(float *w, float *x,
            float *b, float *y,
            int out_dim, int in_dim)
{
  for (int i = 0; i < out_dim; i++) {
    float sum = b[i];
    for (int j = 0; j < in_dim; j++)
      sum += w[i*in_dim + j] * x[j];
    y[i] = sum;
  }
}
\end{lstlisting}
这是整个推理的\textbf{计算核心}。\\
两层循环：外层遍历输出神经元，内层做点积。
\end{block}

\column{0.5\textwidth}
\begin{block}{ReLU + Argmax}
\begin{lstlisting}[basicstyle=\ttfamily\footnotesize, frame=none, numbers=none, backgroundcolor=\color{codebg}]
void relu(float *x, int n) {
  for (int i = 0; i < n; i++)
    if (x[i] < 0.0f) x[i] = 0.0f;
}

int argmax(float *x, int n) {
  int best = 0;
  for (int i = 1; i < n; i++)
    if (x[i] > x[best]) best = i;
  return best;
}
\end{lstlisting}
\textbf{ReLU:} 遍历数组，负数置零。一行代码。

\textbf{Argmax:} 遍历数组，找最大值索引。两行代码。
\end{block}
\end{columns}

\vspace{0.15cm}

\begin{block}{完整前向传播（\texttt{main} 函数中）}
\begin{lstlisting}[basicstyle=\ttfamily\small, frame=none, numbers=none, backgroundcolor=\color{codebg}]
linear(w1, input, b1, h1, 128, 784);  relu(h1, 128);   // 第1层: 784 -> 128
linear(w2, h1,   b2, h2, 64, 128);    relu(h2, 64);    // 第2层: 128 -> 64
linear(w3, h2,   b3, logits, 10, 64);                  // 第3层: 64 -> 10 (无ReLU)
int pred = argmax(logits, 10);                          // 取最大分数为预测类别
\end{lstlisting}
\end{block}
\end{frame}

% ── 4.9 训练优化技巧 ──
\begin{frame}[fragile,shrink=15]{如何进一步优化？——五项实用技巧}
\small

\begin{columns}[T]
\column{0.5\textwidth}
\begin{block}{1. 学习率调度 (LR Schedule)}
\footnotesize
\begin{lstlisting}[basicstyle=\ttfamily\footnotesize, frame=none, numbers=none, backgroundcolor=\color{codebg}]
scheduler = ReduceLROnPlateau(
    optimizer, mode='max',
    patience=2, factor=0.5)
# 每个 epoch 后调用
scheduler.step(test_acc)
\end{lstlisting}
验证准确率不再提升时自动降低学习率。\\
\textbf{原理：} 训练后期用更小步长精细搜索。
\end{block}

\begin{block}{2. 早停 (Early Stopping)}
\footnotesize
当验证准确率连续 N 轮不提升时停止训练，防止过拟合。\\
\textbf{实践：} patience=3，保存最佳模型。
\end{block}

\begin{block}{3. Dropout 正则化}
\footnotesize
训练时随机"丢弃"一部分神经元（如 25--50\%），迫使网络不过度依赖特定神经元。\\
\textbf{代码：} \texttt{nn.Dropout(0.25)}
\end{block}

\column{0.5\textwidth}
\begin{block}{4. 数据增强 (Data Augmentation)}
\footnotesize
\begin{lstlisting}[basicstyle=\ttfamily\footnotesize, frame=none, numbers=none, backgroundcolor=\color{codebg}]
transform = transforms.Compose([
    transforms.RandomRotation(10),
    transforms.RandomAffine(0,
        translate=(0.1, 0.1)),
    transforms.ToTensor(),
])
\end{lstlisting}
随机旋转/平移/缩放训练图片——\textbf{模型永远看到"新图片"}，强制学习不变性。\\
\textbf{效果：} 对 Fashion-MNIST 和 CIFAR-10 提升最显著（+3--8\%）。
\end{block}

\begin{block}{5. 更深的网络}
\footnotesize
当前 CNN 仅 2 层卷积。对于 CIFAR-10：\\
$\to$ 增加卷积层（3--4 层）\\
$\to$ 增加卷积核数（16$\to$32$\to$64$\to$128）\\
$\to$ 引入 BatchNorm 加速收敛
\end{block}
\end{columns}
\end{frame}

% ── 4.10 泛化差距：定义与直观 ──
\begin{frame}[shrink=25]{泛化差距：模型学到的到底是"规律"还是"答案"？}
\small

\begin{columns}[T]
\column{0.55\textwidth}
\begin{block}{泛化差距的正式定义}
\footnotesize
\textbf{泛化 (Generalization)}：模型在训练时未见过的数据上的表现能力。

\vspace{0.05cm}
\textbf{泛化差距 (Generalization Gap)}：
\[
\text{泛化差距} = \text{训练准确率} - \text{测试准确率}
\]

\vspace{-0.1cm}
\begin{itemize}
  \item \textbf{差距小}（$<$2\%）：模型学到的模式在新数据上也成立——泛化好
  \item \textbf{差距大}（$>$5\%）：训练集上高分靠的是"背诵"，新题一做就露馅——过拟合
\end{itemize}
\end{block}

\vspace{0.05cm}

\begin{block}{直观类比：学生考试}
\footnotesize
\begin{center}
\begin{tikzpicture}[node distance=0.4cm]
  \node[draw, rounded corners, fill=lightblue!40, minimum width=6cm, minimum height=0.65cm, font=\footnotesize, align=center] (a)
    {\textbf{场景 1：做过原题} — 考试题和作业题一模一样。\\
     学生只需背答案就能考 99 分。这不代表他"真懂"。};
  \node[draw, rounded corners, fill=redaccent!15, minimum width=6cm, minimum height=0.65cm, font=\footnotesize, align=center, below=of a] (b)
    {\textbf{场景 2：新题测验} — 考试题换了数据，但知识点相同。\\
     背答案的学生分数骤降；真正理解的学生依然高分。};
\end{tikzpicture}
\end{center}
\begin{itemize}
  \item 训练集 = 作业题（学生反复练习过）
  \item 测试集 = 新题（从未见过，但考察同一知识点）
  \item \textbf{泛化差距 = 作业成绩 $-$ 考试成绩}——差距大说明只会做原题
\end{itemize}
\end{block}

\column{0.45\textwidth}
\begin{block}{我们实验中的泛化差距}
\footnotesize
\centering
\begin{tabular}{l c c c}
\toprule
\textbf{数据集} & \textbf{训练 Acc} & \textbf{测试 Acc} & \textbf{差距} \\
\midrule
MNIST MLP    & 99.3\% & 97.4\% & 1.9\% \\
MNIST CNN    & 99.3\% & 98.9\% & 0.4\% \\
Fashion MLP  & 90.8\% & 88.0\% & 2.8\% \\
Fashion CNN  & 92.3\% & 90.3\% & 2.1\% \\
KMNIST CNN   & 98.6\% & 92.3\% & \alert{6.4\%} \\
CIFAR-10 CNN & 72.5\% & 66.8\% & 5.7\% \\
\bottomrule
\end{tabular}
\end{block}

\vspace{0.05cm}

\begin{exampleblock}{关键观察}
\footnotesize
\begin{itemize}
  \item 泛化差距随\textbf{数据复杂度}递增而增大
  \item MNIST 差距 0.4--1.9\% $\to$ 模型确实学到了
  \item KMNIST 差距 6.4\% $\to$ \textbf{严重过拟合}：训练 Acc 很高但泛化很差
  \item CIFAR-10 训练 Acc 才 72.5\% $\to$ \textbf{欠拟合}：连训练集都没学好
\end{itemize}
\end{exampleblock}
\end{columns}
\end{frame}

% ── 4.11 三种诊断与 MNIST 的虚假安全感 ──
\begin{frame}[shrink=20]{三种诊断：从泛化差距判断模型处于什么状态}
\small

\begin{columns}[T]
\column{0.5\textwidth}
\begin{block}{泛化差距的三种诊断}
\footnotesize
\centering
\begin{tabular}{c c c l}
\toprule
训练 Acc & 测试 Acc & 泛化差距 & 诊断 \\
\midrule
高 & 高 & 小 $(<2\%)$ & \textbf{泛化良好} $\checkmark$ \\
高 & 低 & 大 $(>5\%)$ & \textbf{过拟合} $\leftarrow$ 最常见 \\
低 & 低 & 小 & \textbf{欠拟合} $\leftarrow$ 模型太弱 \\
\bottomrule
\end{tabular}

\vspace{0.1cm}
\textbf{诊断逻辑（按顺序）：}
\begin{enumerate}
  \item \textbf{先看训练准确率：}够高吗？不够 $\to$ 欠拟合，先解决容量/优化问题
  \item \textbf{再看泛化差距：}差距大吗？大 $\to$ 过拟合，需要正则化
  \item 训练高 + 差距小 = \textbf{成功}，可以尝试更难的测试
\end{enumerate}
\end{block}

\vspace{0.1cm}

\begin{alertblock}{核心认知}
\footnotesize
\textbf{训练准确率不能代表模型能力。} 泛化差距才是衡量模型价值的真正指标。\\
一个好的机器学习工程师关注测试集表现和泛化差距，而非只看训练集数字。
\end{alertblock}

\column{0.5\textwidth}
\begin{block}{为什么 MNIST 的 99\% 是"虚假的安全感"}
\footnotesize
\begin{itemize}
  \item MNIST 背景纯黑，数字居中，类间差异大——\textbf{数据本身太简单}
  \item 模型只需检测"哪里有白色像素"就能做对大部分题
  \item 同样的 MLP：MNIST 97.4\% $\to$ Fashion-MNIST 88.0\%（骤降 9.4\%）
  \item \textbf{模型从未真正"理解"图像，} 只学会了 MNIST 特定的像素分布
\end{itemize}
\end{block}

\vspace{0.1cm}

\begin{exampleblock}{如何正确评估模型能力？}
\footnotesize
\begin{itemize}
  \item 永远在模型\textbf{从未见过}的数据上报告最终成绩
  \item 使用\textbf{多个不同分布}的测试集（MNIST + Fashion + KMNIST + CIFAR）
  \item 泛化差距 + 跨数据集表现 = 模型能力的完整画像
  \item MNIST 只能作为入门的"Hello World"，不能作为能力的唯一证明
\end{itemize}
\end{exampleblock}
\end{columns}
\end{frame}

% ── 4.12 过拟合与欠拟合 ── (编号接续 4.10/4.11)
\begin{frame}[shrink=20]{过拟合 vs 欠拟合：两个方向的失败模式}
\small

\begin{columns}[T]
\column{0.48\textwidth}
\begin{block}{过拟合 (Overfitting)}
\footnotesize
\textbf{表现：} 训练准确率 $\gg$ 测试准确率（泛化差距大）

\textbf{原因：}
\begin{itemize}
  \item 模型容量过大，记住了训练样本的噪声
  \item 训练数据不足或缺乏多样性
  \item 训练轮数过多
\end{itemize}

\textbf{我们的案例：}
\begin{itemize}
  \item KMNIST CNN: Train 98.6\% vs Test 92.3\%（差距 6.4\%）
  \item 模型在"背诵"假名字形而非学习笔画
\end{itemize}

\textbf{解决方案：}
\begin{enumerate}
  \item 数据增强（增加训练样本多样性）
  \item Dropout 正则化
  \item 早停（Early Stopping）
  \item 减少模型参数
\end{enumerate}
\end{block}

\column{0.52\textwidth}
\begin{block}{欠拟合 (Underfitting)}
\footnotesize
\textbf{表现：} 训练准确率本身就很低

\textbf{原因：}
\begin{itemize}
  \item 模型容量不足，无法捕捉数据中的模式
  \item 优化不充分（学习率太低、训练轮数不够）
  \item 特征表示不够（如用 MLP 处理 CIFAR-10）
\end{itemize}

\textbf{我们的案例：}
\begin{itemize}
  \item CIFAR-10 CNN: Train 72.5\%（训练本身就不高）
  \item 2 层卷积 + 28$\times$28 Resize 不足以处理彩色自然图像
\end{itemize}

\textbf{解决方案：}
\begin{enumerate}
  \item 增加模型容量（更多层、更宽）
  \item 使用更强的模型架构
  \item 减小 Resize 的比例（或直接用 32$\times$32）
  \item 调整学习率
\end{enumerate}
\end{block}
\end{columns}

\vspace{0.1cm}
\footnotesize
\begin{center}
\textbf{关键洞察：} 过拟合和欠拟合需要相反的解决方案。诊断的第一步永远是：\textbf{先看训练准确率本身是否够高。}
\end{center}
\end{frame}

% ── 4.13 调优方法论 ──
\begin{frame}[shrink=20]{调优方法论：系统性地提升模型表现}
\small

\begin{columns}[T]
\column{0.55\textwidth}
\begin{block}{调试优先级（从高到低）}
\footnotesize
\begin{enumerate}
  \item \textbf{先确认代码正确} —— 数据加载、标签对齐、模型定义
  \item \textbf{过拟合一个小批次} —— 取 1--2 个 batch，训练到 loss $\approx$ 0，验证模型能"记住"数据
  \item \textbf{选一个合理的 baseline} —— 用已知能工作的配置（如 lr=1e-3, Adam）
  \item \textbf{调整学习率} —— 最重要的超参数。过大不收敛，过小收敛慢
  \item \textbf{增加训练轮数} —— 如果 loss 仍在下降，继续训练
  \item \textbf{正则化} —— 如果训练准确率远高于测试，加 Dropout / 数据增强
  \item \textbf{调模型结构} —— 加层、加宽、换架构
  \item \textbf{调其他超参数} —— batch\_size, optimizer（Adam $\to$ SGD+Momentum）
\end{enumerate}
\end{block}

\column{0.45\textwidth}
\begin{block}{学习率：最重要的超参数}
\footnotesize
\begin{center}
\begin{tikzpicture}[scale=0.7]
  \draw[->] (0,0) -- (6,0) node[right] {epoch};
  \draw[->] (0,0) -- (0,3.5) node[above] {loss};

  % too high: oscillates
  \draw[redaccent, thick, ->] (0,2.5) -- (2,2.8);
  \draw[redaccent, thick, ->] (2,2.8) -- (4,0.5);
  \draw[redaccent, thick, ->] (4,0.5) -- (5,3.0);
  \node[redaccent, font=\footnotesize] at (5.5,3.0) {太大};

  % too low: slow
  \draw[gray, thick] (0,2.0) -- (5,1.5);
  \node[gray, font=\footnotesize] at (5.5,1.5) {太小};

  % just right
  \draw[mainblue, ultra thick] (0,2.2) .. controls (2,1.8) and (4,0.6) .. (5.5,0.3);
  \node[mainblue, font=\footnotesize] at (5.5,0.0) {合适};
\end{tikzpicture}
\end{center}

\textbf{经验法则：}
\begin{itemize}
  \item Adam: 从 1e-3 开始，尝试 5e-4, 2e-3
  \item SGD: 从 1e-2 开始，尝试 1e-3, 1e-1
  \item 训练 loss 震荡 $\to$ 减小 lr
  \item 训练 loss 下降极慢 $\to$ 增大 lr
\end{itemize}
\end{block}
\end{columns}

\vspace{0.1cm}

\begin{exampleblock}{黄金法则}
\footnotesize
\textbf{一次只改一个变量。} 同时改学习率、模型结构和正则化，你永远不知道哪个改动起了作用。\\
\textbf{记录每次实验结果。} 这正是 \texttt{experiments.md} 的作用——参数和结果是调优的"实验日志"。
\end{exampleblock}
\end{frame}

% ── 4.14 阶段 4 总结 ──
\begin{frame}[shrink=22]{阶段 4 总结：从 MNIST 到泛化的全景认知}
\small

\begin{columns}[T]
\column{0.48\textwidth}
\begin{block}{五个核心收获}
\footnotesize
\begin{enumerate}
  \item \textbf{同一模型，不同表现}\\
  MLP: MNIST 97\% $\to$ Fashion-MNIST 88\%\\
  模型学到的不是通用"视觉智能"

  \item \textbf{CNN 的归纳偏置始终有效}\\
  在所有数据集上 CNN $>$ MLP，\\
  任务越难，CNN 优势越大

  \item \textbf{泛化差距随数据复杂度递增}\\
  MNIST(1.9\%) $\to$ Fashion(2.8\%) $\to$ CIFAR(5.7\%)

  \item \textbf{过拟合与欠拟合需要区分}\\
  训练 Acc 高 + 测试 Acc 低 = 过拟合\\
  训练 Acc 低 = 欠拟合 $\to$ 不同的解决方案

  \item \textbf{调优需要方法论}\\
  一次一个变量，先 baseline，再正则化
\end{enumerate}
\end{block}

\column{0.52\textwidth}
\begin{block}{训练迁移的技术总结}
\footnotesize
\textbf{不变的：}
\begin{itemize}
  \item 训练循环（5 步流程）
  \item 损失函数（CrossEntropyLoss）
  \item 优化器（Adam）
  \item 评估方法（accuracy）
\end{itemize}

\textbf{可变的：}
\begin{itemize}
  \item 数据加载路径（\texttt{DATASET\_INFO}）
  \item 模型输入参数（in\_channels / input\_dim）
  \item 预处理（Resize for CIFAR-10）
  \item 超参数（lr, epochs, batch\_size）
\end{itemize}

\textbf{软件工程视角：}\\
通过参数化隔离变化，\\
用 \texttt{--dataset} 和 \texttt{--model} 切换。
\end{block}
\end{columns}

% ── 5.2 PCA 原理 A ──
\begin{frame}[shrink=18]{PCA（一）：核心思想与数学推导}
\small

\begin{columns}[T]
\column{0.5\textwidth}
\begin{block}{PCA 的直觉}
\footnotesize
\textbf{PCA (Principal Component Analysis)}——主成分分析。

将 784 维像素视为 784 维空间中的点云。核心问题：\textbf{哪个方向上数据"散得最开"？}

\begin{itemize}
  \item 方差越大的方向 = 包含信息越多
  \item PC1 = 方差最大的方向，PC2 = 与 PC1 正交且方差次大的方向
  \item 每个主成分 = 原始特征的线性组合
  \item 降维 = 旋转坐标系，使新轴对齐数据方差方向
\end{itemize}

\textbf{优化视角：} 寻找 $k$ 维线性子空间，最大化投影方差 $\leftrightarrow$ 最小化重构误差 $\|X - Z V_k^T\|^2$。
\end{block}

\column{0.5\textwidth}
\begin{block}{数学推导}
\footnotesize
\textbf{输入：} $X \in \mathbb{R}^{N \times d}$（已中心化）

\textbf{步骤：}
\begin{enumerate}
  \item 协方差矩阵：$\Sigma = \frac{1}{N}X^T X$（$d \times d$ 对称）
  \item 特征分解：$\Sigma = V \Lambda V^T$
  \item $\lambda_1 \ge \dots \ge \lambda_d$ = 各方差的方差
  \item $v_1, \dots, v_d$ = 主成分方向（互相正交）
  \item 取前 $k$ 个：$Z = X V_k$（$N \times k$）
\end{enumerate}

\textbf{本质：} 协方差矩阵的\textbf{特征分解}。特征值 = 该方向的方差，特征向量 = 该方向。
\end{block}

\vspace{0.1cm}

\begin{block}{协方差矩阵 $\Sigma$ 的含义}
\footnotesize
$\Sigma_{ij}$ = 像素 $i$ 和 $j$ 的协方差。Fashion-MNIST 中 $\Sigma$ 是 $784\times784$：
\begin{itemize}
  \item 对角线：单个像素的方差。非对角线：两个像素的联动模式
  \item $v_1$ 是 784 维向量，reshape 回 $28\times28$——\textbf{像一张特征图}
\end{itemize}
\end{block}
\end{columns}
\end{frame}

% ── 5.2b PCA 原理 B ──
% ── 5.2b PCA 原理 B ──
\begin{frame}[shrink=10]{PCA（二）：线性性质、优点与局限}
\small

\begin{columns}[T]
\column{0.5\textwidth}
\begin{block}{PCA 为什么是线性的？}
\footnotesize
每个主成分 = 原始特征的\textbf{线性组合}：
\[
z_1 = v_{1,1}x_1 + v_{1,2}x_2 + \dots + v_{1,784}x_{784}
\]

\textbf{几何理解：} 在 784D 空间中旋转坐标系。新轴 = 主成分方向。降维 = 丢弃方差最小的轴。

\textbf{类比：} 拍摄一个倾斜摆放的物体，旋转相机使其"最宽的面"正对镜头——那就是 PC1。
\end{block}

\vspace{0.1cm}

\begin{block}{PCA 的优点}
\footnotesize
\begin{itemize}
  \item \textbf{计算快：} 只需一次特征分解（$O(d^3)$，$d=784$ 时很快）
  \item \textbf{可解释：} 每个主成分可 reshape 为 $28\times28$ 图片——看到"PCA 学到的特征"
  \item \textbf{确定性强：} 给定数据，PCA 结果唯一（不像神经网络依赖初始化和优化）
  \item \textbf{无超参数：} 只需指定目标维度 $k$
\end{itemize}
\end{block}

\column{0.5\textwidth}
\begin{block}{PCA 的局限}
\footnotesize
\begin{itemize}
  \item \textbf{仅线性：} 数据在曲面上 $\to$ PCA 沿切线方向（丢失弯曲结构）
  \item \textbf{全局性：} 假设数据有一个统一的低维子空间——不适用于数据分片在不同子空间的情况
  \item \textbf{方差 $\neq$ 信息：} 高方差方向可能对应噪声而非语义信息
  \item \textbf{需要中心化：} 数据必须先减去均值，否则第一个主成分可能只捕获"中心位置"
\end{itemize}
\end{block}

\vspace{0.1cm}

\begin{exampleblock}{为什么 PCA 局限在 Fashion-MNIST 上体现不明显？}
\footnotesize
Fashion-MNIST 的变异主要由线性结构主导（前 32 维占 83\%）。这解释了为什么 AE 只比 PCA 高 1.2\%——非线性优势在更复杂的数据（CIFAR-10、自然图像）上才会充分展现。
\end{exampleblock}
\end{columns}
\end{frame}

% ── 5.4 K-means 原理 ──
\begin{frame}[shrink=15]{K-means（一）：最经典的聚类算法——原理与局限}
\small

\begin{columns}[T]
\column{0.5\textwidth}
\begin{block}{K-means 要解决什么问题？}
\footnotesize
\textbf{给定：} $N$ 个数据点 $x_1, \dots, x_N \in \mathbb{R}^d$，指定分成 $K$ 组。

\textbf{目标：} 找到 $K$ 个聚类中心 $\mu_1, \dots, \mu_K$ 和分配方案，最小化\textbf{组内平方和 (WCSS)}：
\[
J = \sum_{k=1}^K \sum_{x \in C_k} \|x - \mu_k\|^2
\]

\textbf{LLoyd 算法（迭代求解）：}
\begin{enumerate}
  \item \textbf{初始化：} 选 $K$ 个初始中心（K-means++ 策略：随机选第一个，后续选离已有中心远的点）
  \item \textbf{分配步 (E-step)：} 每个点分配给最近的中心
  \item \textbf{更新步 (M-step)：} 每个中心移到所属点的均值
  \item 重复 2--3，直到中心不再移动（或达最大迭代次数）
\end{enumerate}

\textbf{关键性质：}
\begin{itemize}
  \item 每步都使 $J$ 减小或不变——保证收敛
  \item 但收敛到\textbf{局部最优}（依赖初始值）——所以多次随机初始化取最佳
\end{itemize}
\end{block}

\column{0.5\textwidth}
\begin{block}{K-means 的局限}
\footnotesize
\textbf{1. 必须预设 $K$：} 需要事先知道聚类数（我们用 10，因为已知 Fashion-MNIST 有 10 类——但在真实无监督场景中不知道）。

\textbf{2. 假设球形簇：} K-means 用欧氏距离，隐含假设每个簇是球形、等大小。如果数据实际呈细长形或不规则形状，K-means 表现差。

\textbf{3. 对离群点敏感：} 一个极端值可以大幅拉偏聚类中心。

\textbf{4. 线性决策边界：} K-means 的簇边界是 Voronoi 图——每个簇由超平面分隔。无法处理嵌套或环形结构。
\end{block}

\vspace{0.1cm}

\begin{block}{为什么在像素空间和 PCA 空间都做？}
\footnotesize
\begin{itemize}
  \item \textbf{原始像素 (784D)：} 维度高，欧氏距离受"维数灾难"影响——高维空间中的距离趋向均匀化
  \item \textbf{PCA 特征 (32D)：} 降噪去冗，但丢失了 PCA 未能捕捉的（非线性）区分信息
  \item \textbf{两者 NMI 接近（0.51 vs 0.49）：} 说明 K-means 的性能瓶颈不在维度，而在其线性/球形假设
\end{itemize}
\end{block}
\end{columns}
\end{frame}

% ── 5.5 K-means 结果 A ──
\begin{frame}[shrink=22]{K-means（二）：聚类评估指标——NMI 与 ARI}
\small

\begin{columns}[T]
\column{0.55\textwidth}
\begin{block}{无监督聚类的评估难题}
\footnotesize
\textbf{核心矛盾：} 无监督学习没有标签——如何判断聚类好坏？

两种评估策略：
\begin{enumerate}
  \item \textbf{内部指标（无需标签）：} 簇内紧密度、簇间分离度（Silhouette Score、Davies-Bouldin Index）
  \item \textbf{外部指标（需真标签）：} 与已知标签的一致性——用于验证、不用于训练
\end{enumerate}

这里我们用外部指标来\textbf{量化评估}，但真标签\textbf{从未参与聚类过程}。
\end{block}

\vspace{0.1cm}

\begin{block}{NMI — 归一化互信息}
\footnotesize
\[
\text{NMI}(U, V) = \frac{I(U; V)}{\sqrt{H(U) \cdot H(V)}}, \quad I(U;V) = \sum_{u,v} P(u,v) \log\frac{P(u,v)}{P(u)P(v)}
\]

\begin{itemize}
  \item $I(U;V)$ = 互信息：知道聚类结果 $U$ 后，对真实标签 $V$ 的不确定性减少多少
  \item $H(U), H(V)$ = 熵：各自的不确定性
  \item 除以 $\sqrt{H(U)H(V)}$ 归一化到 $[0,1]$
  \item \textbf{直觉：} NMI=1 表示聚类和真标签完全一致；NMI$\approx$0 表示几乎独立
\end{itemize}
\end{block}

\column{0.45\textwidth}
\begin{block}{ARI — 调整兰德指数}
\footnotesize
\textbf{兰德指数 (RI)：} 统计所有样本对中，聚类和真标签"意见一致"的比例。

\textbf{问题：} 随机聚类也能得到不错的 RI（如两类各 50\%，随机猜就有 50\% 一致）。

\textbf{ARI 修正了这一点：}
\[
\text{ARI} = \frac{\text{RI} - E[\text{RI}]}{\max(\text{RI}) - E[\text{RI}]}
\]
\begin{itemize}
  \item 减去随机期望值，除以最大可能提升
  \item $<0$ = 比随机还差，$0$ = 随机水平，$1$ = 完美
  \item \textbf{ARI 是更诚实的指标}——不会被类别不均衡"欺骗"
\end{itemize}
\end{block}

\vspace{0.1cm}

\begin{block}{混淆矩阵读法指南}
\footnotesize
每行 = 真实类别，每列 = 聚类簇。\\
格值 = 该类被分到该簇的\textbf{比例}（行归一化）。\\
\textbf{理想：} 对角线全 1.0。\textbf{混乱：} 一类散到多簇。
\end{block}
\end{columns}
\end{frame}

\column{0.52\textwidth}
\begin{block}{混淆矩阵读法}
\footnotesize
行 = 真实类别，列 = 聚类簇。格值 = 该类分配到该簇的比例（行归一化）。
\begin{itemize}
  \item \textbf{理想：} 每行恰好一个格子为 1.0（对角线上）
  \item \textbf{混乱：} 一行散到多个簇——聚类无法统一该类的内部变异
\end{itemize}

\end{block}

% ── 5.6 Autoencoder A ──
\begin{frame}[shrink=18]{Autoencoder（一）：结构、原理与 PCA 对比}
\small

\begin{columns}[T]
\column{0.48\textwidth}
\begin{block}{AE = Encoder + Bottleneck + Decoder}
\footnotesize
\textbf{Encoder：} $784 \to 256 \to 64 \to 32$\\
\textbf{Decoder：} $32 \to 64 \to 256 \to 784$\\
\textbf{Loss：} MSE(原图, 重构图)

\begin{itemize}

  \item Linear Probe：AE 82.25\% vs PCA 81.06\%（+1.2\%）
  \item 增益有限 $\to$ Fashion-MNIST 以线性变异为主。复杂数据上 AE 优势更显著

\small

\begin{columns}[T]
\column{0.5\textwidth}
\begin{block}{无监督训练循环}
\footnotesize
与监督学习完全相同的 5 步——唯一区别：\textbf{标签被丢弃}。

\begin{lstlisting}[basicstyle=\ttfamily\scriptsize, frame=none, numbers=none, backgroundcolor=\color{codebg}]
for images, _ in train_loader:  # 忽略标签
    optimizer.zero_grad()
    recon, z = model(images)

             images.flatten(1))

    optimizer.step()
\end{lstlisting}
\vspace{-0.1cm}
\texttt{for images, \_ in loader}——标签被 \texttt{\_} 丢弃。图片自身既是输入也是监督信号。
\end{block}

\column{0.5\textwidth}
\begin{block}{Bottleneck 大小的权衡}
\footnotesize
\centering
\begin{tabular}{c c c}
\toprule
\textbf{latent\_dim} & \textbf{重构质量} & \textbf{泛化能力} \\
\midrule
2  & 差 & 好（强正则化） \\
32 & 较好（MSE 0.009） & 较好 \\
128 & 很好 & 可能恒等映射 \\
784 & 完美 & 无意义（无压缩） \\
\bottomrule
\end{tabular}
\vspace{0.1cm}

\textbf{直觉：} bottleneck 太窄 $\to$ 信息不够。太宽 $\to$ 网络偷懒学"恒等映射"（直接复制输入），latent 毫无意义。32 维在重构和压缩之间取得平衡。
\end{block}

\vspace{0.1cm}

\begin{exampleblock}{无监督 Loss 的本质}
\footnotesize
监督学习 loss = criterion(pred, label) —— 需要人工标注。\\
无监督 AE loss = MSE(recon, input) —— \textbf{输入本身就是"标签"}。\\
这种"自监督信号"是无监督学习的核心设计模式。
\end{exampleblock}

% ── 5.7c DAE C ──
\begin{frame}[shrink=10]{Denoising AE（三）：鲁棒性直觉与自监督设计}
\small

\begin{columns}[T]
\column{0.5\textwidth}
\begin{alertblock}{为什么 DAE 更鲁棒？}
\footnotesize
\textbf{好老师不考原题，考变体。} 每张训练图加不同噪声 $\to$ 模型被迫忽略表面扰动，提取本质特征（形状 > 纹理）。

\textbf{预期效果：}
\begin{itemize}
  \item latent 特征对输入扰动更不敏感
  \item 学到的表示更接近数据的"不变性"
  \item Linear Probe 可能比 AE 更高——即使 MSE 更差
\end{itemize}

\textbf{重构误差 $\neq$ 表征质量}——这是无监督学习评估中最容易犯的错误。
\end{alertblock}

\column{0.5\textwidth}
\begin{block}{自监督信号设计}
\footnotesize
DAE 是\textbf{自监督学习 (Self-Supervised Learning)} 的经典案例：

\begin{enumerate}
  \item \textbf{不需要人工标签}——pretext task 自动生成监督信号
  \item \textbf{破坏-重建范式}——加噪声破坏输入，要求还原原始信号
  \item \textbf{模型结构零改动}——与 AE 完全相同
  \item \textbf{目标决定表征质量}——同样的结构，不同 pretext task 学到的特征完全不同
\end{enumerate}

这是当代大模型（BERT 的 masked LM、GPT 的 next-token prediction）的核心思想雏形。
\end{block}
\end{columns}
\end{frame}

% ── 5.8 VAE ──
\begin{frame}[shrink=22]{VAE（一）：从确定性编码到概率分布}
\small

\begin{columns}[T]
\column{0.5\textwidth}
\begin{block}{普通 AE vs VAE}
\footnotesize
\textbf{普通 AE：} encoder 输出一个确定的 latent 向量 $z \in \mathbb{R}^{32}$。\\
\textbf{问题：} latent 空间是零散的——只在训练样本附近有值。两个样本之间的 latent 可能对应无意义的输出。

\textbf{VAE：} encoder 输出两个向量：
\begin{itemize}
  \item $\mu \in \mathbb{R}^{32}$——latent 的中心位置
  \item $\log\sigma^2 \in \mathbb{R}^{32}$——latent 的不确定范围
\end{itemize}

\textbf{不是学一个点，而是学一个分布 $\mathcal{N}(\mu, \sigma^2)$！}
\end{block}

\vspace{0.1cm}

\begin{block}{重参数化技巧 (Reparameterization Trick)}
\footnotesize
\textbf{问题：} $z \sim \mathcal{N}(\mu, \sigma^2)$ 这个采样操作不可导——梯度无法流过随机节点。

\textbf{技巧：} 将随机性与参数分离：
\[
z = \mu + \epsilon \cdot e^{\log\sigma^2/2}, \quad \epsilon \sim \mathcal{N}(0, I)
\]
$\epsilon$ 是纯噪声（无参数），梯度只流经 $\mu$ 和 $\log\sigma^2$。PyTorch autograd 可以正常反向传播。
\end{block}

\column{0.5\textwidth}
\begin{block}{VAE 的 Loss 函数}
\footnotesize
\[
\mathcal{L} = \underbrace{\text{MSE}(x, \hat{x})}_{\text{重构项}} + \beta \cdot \underbrace{D_{\text{KL}}\big(\mathcal{N}(\mu,\sigma^2) \| \mathcal{N}(0,I)\big)}_{\text{KL 正则项}}
\]

\textbf{重构项：} 与普通 AE 相同——还原输入。

\textbf{KL 正则项：} 惩罚 latent 分布偏离标准正态 $\mathcal{N}(0,I)$。
\[
D_{\text{KL}} = -\frac{1}{2}\sum_{j=1}^{32}\big(1 + \log\sigma_j^2 - \mu_j^2 - \sigma_j^2\big)
\]

$\beta$ 控制权衡——$\beta$ 小 = 优先重构质量，$\beta$ 大 = 优先 latent 规整性。
\end{block}

\vspace{0.1cm}

\begin{exampleblock}{直观理解}
\footnotesize
普通 AE = 记住每张图的压缩码。VAE = 给每张图分配一个"区域"（均值 + 范围），并且要求所有区域聚拢在原点周围。采样时从原点附近随机取点，decoder 能生成有意义的图片。
\end{exampleblock}
\end{columns}
\end{frame}

% ── 5.8b VAE 原理 B ──
\begin{frame}[shrink=20]{VAE（二）：为什么能生成？——Latent 空间的连续性与 $\beta$ 权衡}
\small

\begin{columns}[T]
\column{0.5\textwidth}
\begin{block}{KL 散度如何"填满" latent 空间}
\footnotesize
\textbf{没有 KL（= 普通 AE）：}
\begin{itemize}
  \item 每个训练样本占据 latent 中一个孤立点
  \item 点之间是"死区"——采样到死区 $\to$ decoder 输出无意义的噪声
\end{itemize}

\textbf{有 KL 约束（= VAE）：}
\begin{itemize}
  \item 每个样本占据一个高斯"气泡"（均值 + 方差）
  \item 所有气泡被拉向原点——latent 空间被"填满"
  \item 采样 $z \sim \mathcal{N}(0,I)$ 大概率落在某个气泡内 $\to$ 生成有意义的图片
\end{itemize}
\end{block}

\vspace{0.1cm}

\begin{block}{$\beta$ 的权衡（$\beta$-VAE）}
\footnotesize
\centering
\begin{tabular}{c c c}
\toprule
$\beta$ 值 & 效果 & 适用场景 \\
\midrule
$\beta < 0.1$ & 重构好，latent 零散 & 需要高质量重构 \\
$\beta = 0.1$ & 平衡（我们的选择） & 兼顾生成与重构 \\
$\beta = 1$   & 标准 VAE & 偏生成 \\
$\beta > 1$   & 强正则，生成多样 & 解纠缠表示学习 \\
\bottomrule
\end{tabular}
\end{block}

\column{0.5\textwidth}
\begin{block}{我们的 Fashion-MNIST VAE 设置}
\footnotesize
\begin{itemize}
  \item $\beta=0.1$——优先重构清晰度
  \item latent\_dim=32（与 AE 一致，便于对比）
  \item 30 epochs, Adam(lr=1e-3), 参数量 442K
  \item Encoder: $784 \to 256 \to 64 \to (\mu, \log\sigma^2)$
  \item Decoder: $32 \to 64 \to 256 \to 784 \to$ Sigmoid
  \item 最佳测试 loss=7.38（= MSE recon + 0.1 $\cdot$ KL）
\end{itemize}
\end{block}

\vspace{0.1cm}

\begin{alertblock}{VAE 的三重身份}
\footnotesize
\textbf{1. 降维模型：} 和 AE 一样，32 维 bottleneck 压缩表示。\\
\textbf{2. 生成模型：} 从 $\mathcal{N}(0,I)$ 采样即可生成新图片——这是 AE 做不到的。\\
\textbf{3. 概率模型：} latent 不是点而是分布——天然支持不确定性量化。
\end{alertblock}
\end{columns}
\end{frame}

% ── 5.10 Linear Probe 对比 ──
\begin{frame}[shrink=18]{Linear Probe（一）：方法——如何公平比较无监督特征？}

\begin{columns}[T]
\column{0.5\textwidth}
\begin{block}{Linear Probe 方法}
\footnotesize
\textbf{问题：} 如何公平比较不同无监督方法学到的特征质量？

\textbf{方案：} 冻结特征提取器，只训练一个\textbf{线性}分类器。

\textbf{步骤：}
\begin{enumerate}
  \item 用无监督方法提取所有图片的特征（不需要标签）
  \item 冻结特征，在上面训练 Logistic Regression
  \item 分类准确率 = 特征质量的代理指标
\end{enumerate}

\textbf{核心逻辑：} 如果\textbf{线性}分类器就能分好类，说明无监督特征已将不同类别自然分开了。分类器越简单，对特征质量要求越高——这是对无监督表征的\textbf{最严格测试}之一。
\end{block}

PCA (32D)       & 81.06\% \\
AE (32D)        & \textbf{82.25\%} \\
\bottomrule
\end{tabular}

AE 略优于 PCA（+1.2\%）。非线性给了有限但实在的增益。

\end{frame}

% ── 5.10b Linear Probe B ──
\begin{frame}[shrink=10]{Linear Probe（二）：解读——为什么 AE 只高 1.2\%？}
\small

\begin{columns}[T]
\column{0.5\textwidth}
\begin{block}{深度分析}
\footnotesize
\textbf{Fashion-MNIST 的变异以线性为主：}
\begin{itemize}
  \item 前 32 维 PCA 已保留 83\% 方差
  \item 剩余的 17\% 中，非线性结构只占一小部分
  \item AE 的非线性优势在 CIFAR-10、人脸等复杂数据上更显著
\end{itemize}

\textbf{Linear Probe 只是单一维度：}
\begin{itemize}
  \item 只测"线性可分性"——AE 的重构能力和 latent 结构未被评估
  \item AE 可以做 PCA 做不到的事：重构可视化、去噪、生成
\end{itemize}
\end{block}

\column{0.5\textwidth}
\begin{alertblock}{核心教训}
\footnotesize
\textbf{1. 简单的 baseline（PCA）往往是强 baseline。} 不要跳过它直接用深度学习。

\textbf{2. 增益大小取决于数据复杂度。} 数据越简单，非线性方法的优势越小。MNIST/Fashion-MNIST 本质是"简单"的。

\textbf{3. 单一指标不够。} Linear Probe 高 $\neq$ 特征完美。AE 的额外价值在于重构、可视化、可扩展性。

\textbf{4. 无监督评估本身是开放问题。} 我们用了真标签做 Linear Probe——但真正的无监督场景没有标签。实践中常用下游任务（分类、聚类）的性能来间接评估。
\end{alertblock}
\end{columns}
\end{frame}

% ── 5.11 五方法全景对比 ──
\begin{frame}[shrink=20]{阶段 5 总结（一）：五种方法全景对比}
\small

\begin{center}
\begin{tabular}{l c c c l}
\toprule
\textbf{方法} & \textbf{标签} & \textbf{参数量} & \textbf{关键指标} & \textbf{一句话} \\
\midrule
PCA          & 无 & 0      & 方差 82.6\% (32D)  & 线性旋转变换，最快 baseline \\
K-means      & 无 & 0      & NMI 0.51 (像素)    & 简单聚类，T恤/衬衫易混淆 \\
AE           & 无 & 440K  & LP 82.3\%, MSE .009 & 非线性压缩，重构清晰 \\
Denoising AE & 无 & 440K  & MSE 0.012          & 从噪声学习，表征更鲁棒 \\
VAE          & 无 & 442K  & loss 7.38          & 可生成新图片，latent 连续 \\
\bottomrule
\end{tabular}
\end{center}

\vspace{0.15cm}

\begin{columns}[T]
\column{0.5\textwidth}
\begin{block}{五个核心收获}
\footnotesize
\begin{enumerate}
  \item \textbf{无监督目标来自数据本身：} 重构、方差最大化、聚类紧密度——不需要人工标签
  \item \textbf{PCA 是必须的 baseline：} 简单但有效，81\% Linear Probe——不要跳过它
  \item \textbf{非线性（AE）带来实在增益：} +1.2\%，在更复杂数据上优势更大
  \item \textbf{去噪 = 鲁棒性：} 牺牲 MSE，换得更鲁棒的特征——重构误差 $\neq$ 表征质量
  \item \textbf{VAE 通向生成：} KL 约束让 latent 连续可采样——从"压缩"到"创造"
\end{enumerate}
\end{block}

\column{0.5\textwidth}
\begin{block}{关键教训}
\footnotesize
\begin{itemize}
  \item \textbf{线性 baseline 不能省：} PCA 81\% vs AE 82\%——80\% 的收益来自线性
  \item \textbf{MSE 不是全部：} DAE 的 MSE 更高但表征更鲁棒——指标取决于目标
  \item \textbf{无监督评估是难题：} 我们用 Linear Probe（外部标签）——实践中需下游任务
  \item \textbf{从重构到生成只需一步：} AE + KL = VAE——结构改变极小，能力跃升巨大
\end{itemize}
\end{block}
\end{columns}
\end{frame}

\footnotesize

\textbf{线性 $\to$ 非线性 $\to$ 鲁棒 $\to$ 概率 $\to$ 生成}
